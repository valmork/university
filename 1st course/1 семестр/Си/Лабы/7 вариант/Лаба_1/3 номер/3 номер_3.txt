#include <stdio.h>
#include <stdlib.h>
int main(){    
    float a, b, h, r;
    char text = 'X';
       for(;; text){
                 puts("---Menu---");

                 puts("T - Trapeciya");
                 puts("R - Pryamougolnik");

                 if (text == 'C'){
                    printf("Vvedite radius: ");

                    printf("%.2f\n\n", r * r * 3.14);     

                    }

                    printf("Vvedite 1 osnovaniye: ");
                    scanf("%f", &a);
                    printf("Vvedite 2 osnovaniye: ");
                    scanf("%f", &b);
                    printf("Vvedite 1 visotu: ");
                    scanf("%f", &h);
                    printf("%.2f\n\n", ((a * b)/2)*h);

                    scanf("%f", &a);

                    }
                 if (text == 'E'){
                      break;
                 }
                 scanf("%c", &text);
    
    }
    return 0;
}
